#include <stdio.h>
#include <stdlib.h>

int board[20][20];
int n;

void printBoard()
{
int i, j;

printf("\nSolution:\n");

for (i = 0; i < n; i++)
{
for (j = 0; j < n; j++)
{
printf("%d ", board[i][j]);
}
printf("\n");
}
}

int isSafe(int row, int col)
{
int i, j;

// Check column
for (i = 0; i < row; i++)
{
if (board[i][col] == 1)
return 0;
}

// Check upper-left diagonal
for (i = row - 1, j = col - 1;
i >= 0 && j >= 0; i--, j--)
{
if (board[i][j] == 1)
return 0;
}

// Check upper-right diagonal
for (i = row - 1, j = col + 1;
i >= 0 && j < n; i--, j++)
{
if (board[i][j] == 1)
return 0;
}

return 1;
}

int solveNQueens(int row)
{
int col;

if (row == n)
return 1;

for (col = 0; col < n; col++)
{
if (isSafe(row, col))
{
board[row][col] = 1;

if (solveNQueens(row + 1))
return 1;

// Backtracking
board[row][col] = 0;
}
}

return 0;
}

int main()
{
int i, j;

printf("Enter the number of Queens (N): ");
scanf("%d", &n);

if (n < 4 || n > 20)
{
printf("Enter N between 4 and 20.\n");
return 0;
}

// Initialize board with zeros
for (i = 0; i < n; i++)
{
for (j = 0; j < n; j++)
{
board[i][j] = 0;
}
}

// Place the first Queen at (0, 0)
board[0][0] = 1;

// Solve for the remaining Queens
if (solveNQueens(1))
{
printBoard();
}
else
{
printf("No solution exists with first Queen at (0,0).\n");
}

return 0;
}
